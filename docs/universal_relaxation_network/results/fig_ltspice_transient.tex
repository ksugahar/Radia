\begin{tikzpicture}

\begin{axis}[
    name=plota,

    xlabel={Time (ms)},

    ymin=-0.1, ymax=1.15,

] {results/ltspice_waveform_data.csv};

% --- Subplot (b): Input current ---
\begin{axis}[
    name=plotb,
    at={(plota.south east)},

\addplot[green!50!black, line width=1pt, no markers] table[

\node[anchor=north west, font=\footnotesize\bfseries] at (rel axis cs:0.02,0.98) {(b)};

\node[fill=green!15, draw=green!60!black, rounded corners=2pt,
      font=\scriptsize, inner sep=2pt]
    at (rel axis cs:0.72,0.15) {No ringing};
\end{axis}
\end{tikzpicture}
\caption{LTspice transient simulation of URN-synthesized RLC ladder circuit.
(a)~Input voltage pulse (PULSE source, 0--1\,V, 5\,ms on, 5\,ms off).
(b)~Input current response showing smooth RC charging without spurious oscillations,
confirming the passivity-by-construction property of the URN circuit synthesis.
\emph{Note: Actual LTspice simulation via PyLTSpice, not analytical approximation.}}
\label{fig:ltspice_transient}
