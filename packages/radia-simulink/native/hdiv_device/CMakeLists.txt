# Plain MEX over the shared nvcc-built HDiv kernel archive; no mexcuda, NGSolve or Python.
cmake_minimum_required(VERSION 3.24)
project(radia_hdiv_device LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT MSVC_VERSION EQUAL 1944)
  message(FATAL_ERROR "Windows MSVC 14.44 baseline required")
endif()
foreach(required RADIA_CUDA_OBJECT RADIA_OUTPUT RADIA_CUDART RADIA_CUBLAS RADIA_CUSOLVER Matlab_ROOT_DIR)
  if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
    message(FATAL_ERROR "Missing explicit ${required}")
  endif()
endforeach()
set(RADIA_HDIV "${CMAKE_CURRENT_SOURCE_DIR}/../../../../src/core/gpu/hdiv")
include("${RADIA_HDIV}/cmake/HDivKernels.cmake")
find_package(Matlab REQUIRED COMPONENTS MX_LIBRARY)
matlab_add_mex(NAME radia_hdiv_device SRC radia_hdiv_device_mex.cpp LINK_TO radia_hdiv_kernels R2018a)
target_compile_features(radia_hdiv_device PRIVATE cxx_std_20)
target_compile_definitions(radia_hdiv_device PRIVATE NOMINMAX)
target_compile_options(radia_hdiv_device PRIVATE /EHsc /MD /O2 /fp:precise /utf-8)
target_include_directories(radia_hdiv_device PRIVATE "${RADIA_HDIV}" "${CMAKE_CURRENT_SOURCE_DIR}/..")
set_target_properties(radia_hdiv_device PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${RADIA_OUTPUT}" LIBRARY_OUTPUT_DIRECTORY "${RADIA_OUTPUT}")
