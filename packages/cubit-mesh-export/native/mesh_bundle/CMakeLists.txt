cmake_minimum_required(VERSION 3.26)
project(CubitNeutralMesh LANGUAGES CXX)
find_package(Cubit REQUIRED CONFIG)
find_package(CubitCGM REQUIRED CONFIG HINTS "${Cubit_DIR}")
add_library(cubit_mesh_bundle MODULE NeutralMesh.cpp)
target_compile_features(cubit_mesh_bundle PRIVATE cxx_std_17)
target_link_libraries(cubit_mesh_bundle PRIVATE cubiti cubit_util cubit_cgmcore)
if(MSVC)
  target_compile_options(cubit_mesh_bundle PRIVATE /W4 /utf-8)
endif()
