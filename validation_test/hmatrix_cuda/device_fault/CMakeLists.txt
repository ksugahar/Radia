# Test-only MEX: the unmodified adapter source and reviewed kernel object with the fault shim. Never shipped.
cmake_minimum_required(VERSION 3.24)
project(radia_hdiv_device_fault LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT MSVC_VERSION EQUAL 1944)
  message(FATAL_ERROR "Windows MSVC 14.44 baseline required")
endif()
foreach(required RADIA_CUDA_OBJECT RADIA_OUTPUT RADIA_CUDART RADIA_CUBLAS RADIA_CUSOLVER Matlab_ROOT_DIR)
  if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
    message(FATAL_ERROR "Missing explicit ${required}")
  endif()
endforeach()
set(RADIA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(RADIA_HDIV "${RADIA_ROOT}/src/core/gpu/hdiv")
set(RADIA_NATIVE "${RADIA_ROOT}/packages/radia-simulink/native")
include("${RADIA_HDIV}/cmake/HDivKernels.cmake")
find_package(Matlab REQUIRED COMPONENTS MX_LIBRARY)
set(adapter "${RADIA_NATIVE}/hdiv_device/radia_hdiv_device_mex.cpp")
matlab_add_mex(NAME radia_hdiv_device_fault SRC "${adapter}" fault_shim.cpp LINK_TO radia_hdiv_kernels R2018a)
set_source_files_properties("${adapter}" PROPERTIES COMPILE_OPTIONS "/FI${CMAKE_CURRENT_SOURCE_DIR}/fault_mex_source.hpp")
target_compile_features(radia_hdiv_device_fault PRIVATE cxx_std_20)
target_compile_definitions(radia_hdiv_device_fault PRIVATE NOMINMAX)
target_compile_options(radia_hdiv_device_fault PRIVATE /EHsc /MD /O2 /fp:precise /utf-8)
target_include_directories(radia_hdiv_device_fault PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${RADIA_HDIV}" "${RADIA_NATIVE}")
set_target_properties(radia_hdiv_device_fault PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${RADIA_OUTPUT}" LIBRARY_OUTPUT_DIRECTORY "${RADIA_OUTPUT}")
