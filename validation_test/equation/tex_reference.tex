% Ask TeX for the exact size of each box it sets, so the comparison is against
% numbers rather than against a picture of numbers.
%
% XeLaTeX with unicode-math and Latin Modern Math, because that is the FONT
% this editor sets in -- the same OpenType file, read through the same MATH
% table.  \usepackage{lmodern} looks like the same typeface and is not: it
% sends TeX to the Type1 lmex10 extension font, whose radicals and large
% operators are built from different pieces at different sizes.  Comparing
% against it reported an 18.5% error in the radical that was the reference's,
% not ours, and a "correction" made from that number had to be undone.
\documentclass[12pt]{article}
\usepackage{amsmath}
\usepackage{unicode-math}
\setmathfont{Latin Modern Math}
\pagestyle{empty}
\begin{document}

\newwrite\out
\immediate\openout\out=dims.txt

% name, then the maths.  wd/ht/dp are width, height above the baseline and
% depth below it, in points.
\def\M#1#2{%
  \setbox0=\hbox{$\displaystyle#2$}%
  \immediate\write\out{#1 \the\wd0\space \the\ht0\space \the\dp0}%
}

\M{x}{x}
\M{ab}{ab}
\M{abc}{abc}

\M{frac}{\frac{a}{b}}
\M{frac_wide}{\frac{abc}{d}}
\M{frac_nested}{\frac{\frac{p}{q}}{c}}
\M{frac_nested2}{\frac{\frac{\frac{1}{2}}{3}}{4}}

\M{sqrt2}{\sqrt{2}}
\M{sqrt_frac}{\sqrt{\frac{a}{b}}}
\M{sqrt_deep}{\sqrt{\frac{\frac{a}{b}}{c}}}
\M{root3}{\sqrt[3]{x}}

\M{sup}{x^{2}}
\M{sub}{x_{i}}
\M{subsup}{x_{i}^{2}}
\M{sup_sup}{x^{y^{z}}}

\M{int}{\int_{0}^{T}}
\M{int_f}{\int_{0}^{T}f}
\M{sum}{\sum_{n=1}^{N}}
\M{oint}{\oint_{C}}

\M{plus}{a+b}
\M{eq}{a=b}
\M{comma}{a,b}
\M{paren}{a(b)}
\M{times}{a*b}
\M{fence}{\left(\frac{1}{2}\right)}

% --- the pieces, so a disagreement can be attributed rather than guessed at --
\M{s2}{\scriptstyle 2}
\M{si}{\scriptstyle i}
\M{sn1}{\scriptstyle n=1}
\M{sN}{\scriptstyle N}
\M{ssz}{\scriptscriptstyle z}
\M{intg}{\int}
\M{sumg}{\sum}
\M{ointg}{\oint}
\M{sqrtg}{\sqrt{\vphantom{2}}}

\immediate\write\out{QUADtext 	heontdimen6	extfont2\space 0pt 0pt}
\immediate\write\out{QUADscript 	heontdimen6\scriptfont2\space 0pt 0pt}
\immediate\write\out{QUADss 	heontdimen6\scriptscriptfont2\space 0pt 0pt}
\immediate\write\out{SCRIPTSPACE 	he\scriptspace\space 0pt 0pt}
\immediate\write\out{NULLDELIM 	he
ulldelimiterspace\space 0pt 0pt}
\mbox{}

% --- landed 2026-08-19: the vector face, prescripts, arrows, one-sided
% fences, the bar down the middle of a bra-ket, and the two fraction
% styles named outright.  \symbfit is the bold ITALIC alphabet, which is
% the face a vector takes here; \mathbf is upright bold and a different
% thing.
\M{bold_italic}{\symbfit{B}}
\M{bold_upright}{\mathbf{B}}
\M{vec_bm}{\vec{\symbfit{B}}}
\M{prescript}{{}^{14}_{6}C}
\M{overrightarrow}{\overrightarrow{AB}}
\M{overleftarrow}{\overleftarrow{AB}}
\M{overset}{\overset{a}{=}}
\M{underset}{\underset{n}{\max}}
\M{xrightarrow}{\xrightarrow{k}}
\M{xrightarrow_both}{\xrightarrow[m]{k}}
\M{braket}{\left\langle \psi \middle| \phi \right\rangle}
\M{evalbar}{\left. f \right|_{a}^{b}}
\M{onesided_brace}{\left\{ x \right.}
\M{dbloverline}{\overline{\overline{x}}}
\M{dblunderline}{\underline{\underline{x}}}
\M{dfrac}{\dfrac{a}{b}}
\M{tfrac}{\tfrac{a}{b}}

\immediate\closeout\out
\end{document}
