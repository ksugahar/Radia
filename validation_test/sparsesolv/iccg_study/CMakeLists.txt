cmake_minimum_required(VERSION 3.20)
project(iccg_study LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
foreach(variant baseline natural hermitian combined)
  pybind11_add_module(iccg_${variant} study.cpp)
  target_compile_features(iccg_${variant} PRIVATE cxx_std_17)
  target_include_directories(iccg_${variant} PRIVATE "${VARIANT_ROOT}/${variant}")
  target_compile_definitions(iccg_${variant} PRIVATE STUDY_MODULE=iccg_${variant})
endforeach()
